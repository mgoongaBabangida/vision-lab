cmake_minimum_required(VERSION 3.23)
project(vision_lab VERSION 0.1.0 LANGUAGES CXX)

option(VISIONLAB_WITH_OPENCV "Enable image/video-file input through OpenCV 4" OFF)
option(VISIONLAB_BUILD_VIEWER "Build the optional SDL2/OpenGL/ImGui viewer" OFF)
set(VISIONLAB_THIRD_PARTY_ROOT "" CACHE PATH "Optional shared dependencies directory (Windows SDL2 layout)")

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)

function(visionlab_target target)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(visionlab_core src/core/image.cpp src/core/pipeline.cpp src/core/synthetic_source.cpp)
add_library(visionlab::core ALIAS visionlab_core)
target_include_directories(visionlab_core PUBLIC include)
visionlab_target(visionlab_core)

if(VISIONLAB_WITH_OPENCV)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
    add_library(visionlab_opencv src/opencv/video_source.cpp src/opencv/image_source.cpp)
    add_library(visionlab::opencv ALIAS visionlab_opencv)
    target_link_libraries(visionlab_opencv PUBLIC visionlab::core PRIVATE ${OpenCV_LIBS})
    target_include_directories(visionlab_opencv SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
    visionlab_target(visionlab_opencv)
endif()

add_library(visionlab_app_common STATIC apps/common/options.cpp apps/common/catalogs.cpp apps/common/viewer_session.cpp)
target_include_directories(visionlab_app_common PUBLIC apps/common)
target_link_libraries(visionlab_app_common PUBLIC visionlab::core)
visionlab_target(visionlab_app_common)
if(VISIONLAB_WITH_OPENCV)
    target_link_libraries(visionlab_app_common PRIVATE visionlab::opencv)
    target_compile_definitions(visionlab_app_common PRIVATE VISIONLAB_WITH_OPENCV=1)
endif()

add_executable(visionlab_cli apps/cli/main.cpp)
target_link_libraries(visionlab_cli PRIVATE visionlab_app_common)
visionlab_target(visionlab_cli)

if(VISIONLAB_BUILD_VIEWER)
    include(cmake/SDL2Dependency.cmake)
    find_package(OpenGL REQUIRED)
    add_library(visionlab_imgui STATIC external/imgui/imgui.cpp external/imgui/imgui_draw.cpp
        external/imgui/imgui_tables.cpp external/imgui/imgui_widgets.cpp
        external/imgui/imgui_impl_sdl.cpp external/imgui/imgui_impl_opengl3.cpp)
    target_include_directories(visionlab_imgui SYSTEM PUBLIC external/imgui)
    target_link_libraries(visionlab_imgui PUBLIC SDL2::SDL2 OpenGL::GL ${CMAKE_DL_LIBS})
    add_executable(visionlab_viewer apps/viewer/main.cpp apps/viewer/viewer_app.cpp apps/viewer/graphics.cpp)
    target_link_libraries(visionlab_viewer PRIVATE visionlab_app_common visionlab_imgui)
    target_compile_definitions(visionlab_viewer PRIVATE SDL_MAIN_HANDLED
        VISIONLAB_DEFAULT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT visionlab_viewer)
    set_target_properties(visionlab_viewer PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    visionlab_target(visionlab_viewer)
    if(WIN32)
        # SDL2's shared imported target must carry the runtime DLL location.
        get_target_property(sdl_type SDL2::SDL2 TYPE)
        if(sdl_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET visionlab_viewer POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:visionlab_viewer>"
                VERBATIM)
        endif()
    endif()
endif()

# Deploy OpenCV's imported runtime DLLs and optional FFmpeg plugin beside Windows apps/tests.
# This makes F5 and direct launches work without editing the user's global PATH.
if(WIN32 AND VISIONLAB_WITH_OPENCV)
    set(visionlab_cv_runtime)
    foreach(cv_target IN LISTS OpenCV_LIBS)
        if(TARGET ${cv_target})
            get_target_property(cv_type ${cv_target} TYPE)
            if(cv_type STREQUAL "SHARED_LIBRARY")
                list(APPEND visionlab_cv_runtime "$<TARGET_FILE:${cv_target}>")
                foreach(cv_config RELEASE DEBUG)
                    get_target_property(cv_location ${cv_target} IMPORTED_LOCATION_${cv_config})
                    if(cv_location)
                        get_filename_component(cv_bin "${cv_location}" DIRECTORY)
                        file(GLOB cv_video_plugins "${cv_bin}/opencv_videoio_ffmpeg*.dll")
                        list(APPEND visionlab_cv_runtime ${cv_video_plugins})
                    endif()
                endforeach()
            endif()
        endif()
    endforeach()
    list(REMOVE_DUPLICATES visionlab_cv_runtime)
    foreach(app visionlab_cli visionlab_viewer visionlab_session_tests visionlab_opencv_tests)
        if(TARGET ${app} AND visionlab_cv_runtime)
            add_custom_command(TARGET ${app} POST_BUILD
                COMMAND ${CMAKE_COMMAND} "-DRUNTIME_FILES=${visionlab_cv_runtime}"
                    "-DDESTINATION=$<TARGET_FILE_DIR:${app}>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CopyRuntime.cmake"
                VERBATIM)
        endif()
    endforeach()
endif()
