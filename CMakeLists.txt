cmake_minimum_required(VERSION 3.23)
project(vision_lab VERSION 0.1.0 LANGUAGES CXX)

option(VISIONLAB_WITH_OPENCV "Enable video-file input through OpenCV 4" OFF)
option(VISIONLAB_BUILD_VIEWER "Build the optional SDL2 viewer" OFF)
set(VISIONLAB_THIRD_PARTY_ROOT "" CACHE PATH "Optional shared dependencies directory (Windows SDL2 layout)")

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)

function(visionlab_target target)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(visionlab_core src/core/image.cpp src/core/pipeline.cpp src/core/synthetic_source.cpp)
add_library(visionlab::core ALIAS visionlab_core)
target_include_directories(visionlab_core PUBLIC include)
visionlab_target(visionlab_core)

if(VISIONLAB_WITH_OPENCV)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc videoio)
    add_library(visionlab_opencv src/opencv/video_source.cpp)
    add_library(visionlab::opencv ALIAS visionlab_opencv)
    target_link_libraries(visionlab_opencv PUBLIC visionlab::core PRIVATE ${OpenCV_LIBS})
    target_include_directories(visionlab_opencv SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
    visionlab_target(visionlab_opencv)
endif()

add_library(visionlab_app_common STATIC apps/common/options.cpp)
target_include_directories(visionlab_app_common PUBLIC apps/common)
target_link_libraries(visionlab_app_common PUBLIC visionlab::core)
visionlab_target(visionlab_app_common)
if(VISIONLAB_WITH_OPENCV)
    target_link_libraries(visionlab_app_common PRIVATE visionlab::opencv)
    target_compile_definitions(visionlab_app_common PRIVATE VISIONLAB_WITH_OPENCV=1)
endif()

add_executable(visionlab_cli apps/cli/main.cpp)
target_link_libraries(visionlab_cli PRIVATE visionlab_app_common)
visionlab_target(visionlab_cli)

if(VISIONLAB_BUILD_VIEWER)
    include(cmake/SDL2Dependency.cmake)
    add_executable(visionlab_viewer apps/viewer/main.cpp)
    target_link_libraries(visionlab_viewer PRIVATE visionlab_app_common SDL2::SDL2)
    target_compile_definitions(visionlab_viewer PRIVATE SDL_MAIN_HANDLED)
    visionlab_target(visionlab_viewer)
    if(WIN32)
        # SDL2's shared imported target must carry the runtime DLL location.
        get_target_property(sdl_type SDL2::SDL2 TYPE)
        if(sdl_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET visionlab_viewer POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:visionlab_viewer>"
                VERBATIM)
        endif()
    endif()
endif()

if(BUILD_TESTING)
    add_executable(visionlab_core_tests tests/core_tests.cpp)
    target_link_libraries(visionlab_core_tests PRIVATE visionlab::core)
    visionlab_target(visionlab_core_tests)
    add_test(NAME core_contracts COMMAND visionlab_core_tests)
    add_test(NAME cli_smoke COMMAND visionlab_cli --frames 3)
    set_tests_properties(cli_smoke PROPERTIES PASS_REGULAR_EXPRESSION "2,0.066667,320,240,0")
    add_test(NAME cli_rejects_zero COMMAND visionlab_cli --frames 0)
    add_test(NAME cli_rejects_unknown_option COMMAND visionlab_cli --unknown)
    set_tests_properties(cli_rejects_zero cli_rejects_unknown_option PROPERTIES WILL_FAIL TRUE)
    if(NOT VISIONLAB_WITH_OPENCV)
        add_test(NAME cli_explains_missing_opencv COMMAND visionlab_cli --input nonexistent.mp4)
        set_tests_properties(cli_explains_missing_opencv PROPERTIES WILL_FAIL TRUE)
    endif()
    if(VISIONLAB_BUILD_VIEWER)
        add_test(NAME viewer_smoke COMMAND visionlab_viewer --frames 3)
        set_tests_properties(viewer_smoke PROPERTIES
            ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 15)
    endif()
    if(VISIONLAB_WITH_OPENCV)
        add_executable(visionlab_opencv_tests tests/opencv_tests.cpp)
        target_link_libraries(visionlab_opencv_tests PRIVATE visionlab::opencv ${OpenCV_LIBS})
        target_include_directories(visionlab_opencv_tests SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
        visionlab_target(visionlab_opencv_tests)
        add_test(NAME opencv_video_contracts COMMAND visionlab_opencv_tests)
    endif()
endif()
